# Direct-democracy votes: automatic, they replace vanilla "Summon the Diet"
# (estate_privileges_and_agendas_events.3, hidden with direct democracy by blocked_call_diet).
# Unlike the vanilla diet, they include the Communes and can be refused.
namespace = hel_diet

country_event = {
	id = hel_diet.2
	title = hel_diet.2.t
	desc = hel_diet.2.d
	picture = NOBLE_ESTATE_DEMANDS_eventPicture
	trigger = {
		hel_is_confederation = yes
		has_estate_privilege = estate_hel_communes_direct_democracy
		NOT = { has_any_active_estate_agenda = yes }
		has_not_any_estate_disaster_active = yes
		hel_diet_off_cooldown = yes
	}
	mean_time_to_happen = { months = 1 }

	immediate = {
		hidden_effect = {
			hel_mark_diet_held = yes
			if = {
				limit = { has_estate = estate_church }
				set_country_flag = hel_church_present_agenda
				generate_estate_agenda = estate_church
			}
			if = {
				limit = { has_estate = estate_burghers }
				set_country_flag = hel_burghers_present_agenda
				generate_estate_agenda = estate_burghers
			}
			if = {
				limit = { has_estate = estate_hel_communes }
				set_country_flag = hel_communes_present_agenda
				generate_estate_agenda = estate_hel_communes
			}
		}
	}

	after = {
		clr_country_flag = hel_church_present_agenda
		clr_country_flag = hel_burghers_present_agenda
		clr_country_flag = hel_communes_present_agenda
		clear_estate_agenda_cache = ROOT
	}

	option = {
		name = estate_privileges_and_agendas_events.3.b
		trigger = { has_country_flag = hel_church_present_agenda }
		start_estate_agenda = estate_church
		ai_chance = { factor = 1 }
	}
	option = {
		name = estate_privileges_and_agendas_events.3.e
		trigger = { has_country_flag = hel_burghers_present_agenda }
		start_estate_agenda = estate_burghers
		ai_chance = { factor = 1 }
	}
	option = {
		name = hel_diet_option_communes
		trigger = { has_country_flag = hel_communes_present_agenda }
		start_estate_agenda = estate_hel_communes
		ai_chance = { factor = 1 }
	}
	option = {
		name = hel_diet.2.refuse
		add_republican_tradition = -5
		if = {
			limit = { has_estate = estate_church }
			add_estate_loyalty = { estate = estate_church loyalty = -5 }
		}
		if = {
			limit = { has_estate = estate_burghers }
			add_estate_loyalty = { estate = estate_burghers loyalty = -5 }
		}
		if = {
			limit = { has_estate = estate_hel_communes }
			add_estate_loyalty = { estate = estate_hel_communes loyalty = -5 }
		}
		ai_chance = { factor = 0 }
	}
}
