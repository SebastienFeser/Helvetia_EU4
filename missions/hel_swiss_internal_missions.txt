# Swiss internal tree (top of the mission window, rows 1-6). Conquest tree goes below (rows 7+).

# Column 1: Confederation
hel_swi_internal_confederation = {
	slot = 1
	generic = no
	ai = yes
	has_country_shield = yes
	potential = {
		tag = SWI
	}

	hel_swi_stans_convenant = {
		icon = consolidate_switzerland
		position = 1
		required_missions = { }
		trigger = {
			stability = 1
			is_at_war = no
		}
		effect = {
			add_stability = 1
			add_country_modifier = {
				name = hel_stans_convenant
				duration = 7300
			}
			# Claims for the first conquest missions (Romandie, Grisons)
			switzerland_area = {
				limit = { NOT = { owned_by = ROOT } NOT = { is_core = ROOT } }
				add_permanent_claim = ROOT
			}
			romandie_area = {
				limit = { NOT = { owned_by = ROOT } NOT = { is_core = ROOT } }
				add_permanent_claim = ROOT
			}
		}
	}

	hel_swi_integrate_cantons = {
		icon = integrate_the_three_leagues
		position = 2
		required_missions = { hel_swi_stans_convenant }
		provinces_to_highlight = {
			area = switzerland_area
			owned_by = ROOT
			OR = {
				NOT = { is_core = ROOT }
				has_owner_accepted_culture = no
			}
		}
		trigger = {
			switzerland_area = {
				type = all
				OR = {
					NOT = { owned_by = ROOT }
					AND = {
						is_core = ROOT
						has_owner_accepted_culture = yes
					}
				}
			}
		}
		effect = {
			switzerland_area = {
				limit = { owned_by = ROOT }
				random_list = {
					1 = { add_base_tax = 1 }
					1 = { add_base_production = 1 }
					1 = { add_base_manpower = 1 }
				}
			}
			custom_tooltip = hel_unlock_cantonal_autonomy_tt
			hidden_effect = { set_country_flag = hel_cantonal_autonomy_unlocked }
		}
	}

	hel_swi_federal_diet = {
		icon = reorganize_the_confederacy
		position = 3
		required_missions = { hel_swi_integrate_cantons }
		trigger = {
			estate_loyalty = { estate = estate_church loyalty = 50 }
			estate_loyalty = { estate = estate_burghers loyalty = 50 }
			estate_loyalty = { estate = estate_hel_communes loyalty = 50 }
		}
		effect = {
			add_adm_power = 50
			add_country_modifier = {
				name = hel_federal_diet
				duration = -1
			}
		}
	}

	hel_swi_confederate_cities = {
		icon = expand_the_confederacy
		position = 4
		required_missions = { hel_swi_federal_diet }
		provinces_to_highlight = {
			owned_by = ROOT
			OR = { area = switzerland_area area = romandie_area }
			NOT = { has_trade_building_trigger = yes }
			NOT = { has_production_building_trigger = yes }
		}
		trigger = {
			num_of_owned_provinces_with = {
				value = 4
				OR = { area = switzerland_area area = romandie_area }
				OR = {
					has_trade_building_trigger = yes
					has_production_building_trigger = yes
				}
			}
		}
		effect = {
			every_owned_province = {
				limit = {
					OR = { area = switzerland_area area = romandie_area }
					OR = {
						has_trade_building_trigger = yes
						has_production_building_trigger = yes
					}
				}
				add_base_production = 1
			}
		}
	}

	hel_swi_thirteen_cantons = {
		icon = switzerland_and_the_empire
		position = 5
		required_missions = { hel_swi_confederate_cities }
		provinces_to_highlight = {
			owned_by = ROOT
			OR = { area = switzerland_area area = romandie_area }
			NOT = { development = 15 }
		}
		trigger = {
			num_of_owned_provinces_with = {
				value = 5
				OR = { area = switzerland_area area = romandie_area }
				development = 15
			}
		}
		effect = {
			add_country_modifier = {
				name = hel_thirteen_cantons
				duration = -1
			}
		}
	}
}
