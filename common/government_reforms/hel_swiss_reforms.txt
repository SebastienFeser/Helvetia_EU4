	icon = "landscape"
	allow_normal_conversion = yes
	# Unlocked by mission hel_swi_integrate_cantons
	potential = {
		hel_is_confederation = yes
		OR = {
			has_country_flag = hel_cantonal_autonomy_unlocked
			have_had_reform = hel_cantonal_autonomy_reform
		}
	}
	modifiers = {
		min_autonomy = 10
		development_cost = -0.15
		global_unrest = -2
		state_maintenance_modifier = -0.25
	}
	# Cancels the non-accepted culture penalty (hel_cantonal_tolerance, attached by events/hel_reform_events.txt)
	effect = {
		hidden_effect = { country_event = { id = hel_reforms.1 } }
	}
	removed_effect = {
		hidden_effect = { country_event = { id = hel_reforms.2 } }
		add_stability = -1
		add_country_modifier = {
			name = hel_cantons_betrayed
			duration = 3650
		}
		every_owned_province = {
			limit = { is_capital = no }
			random = {
				chance = 25
				spawn_rebels = {
					type = nationalist_rebels
					size = 1
				}
			}
		}
	}
	ai = {
		factor = 1
	}
}
