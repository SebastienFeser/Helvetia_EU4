# Merged with vanilla on_actions (other mods rely on this, e.g. Advisor Types Expanded).

on_province_owner_change = {
	if = {
		limit = {
			owner = { has_reform = hel_cantonal_autonomy_reform }
		}
		remove_province_triggered_modifier = hel_cantonal_tolerance
		add_province_triggered_modifier = hel_cantonal_tolerance
	}
}

# No nobility estate for Switzerland (same flag as vanilla disable_estate_effect)
on_startup = {
	if = {
		limit = { tag = SWI }
		set_country_flag = disables_estate_nobles_flag
	}
}
