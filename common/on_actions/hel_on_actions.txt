# Merged with vanilla on_actions (other mods rely on this, e.g. Advisor Types Expanded).

on_province_owner_change = {
	if = {
		limit = {
			owner = { has_reform = hel_cantonal_autonomy_reform }
		}
		remove_province_triggered_modifier = hel_cantonal_tolerance
		add_province_triggered_modifier = hel_cantonal_tolerance
	}
}

# Peace chronometer for mission hel_swi_recognized_neutrality: flag date = start of current peace (any war resets it)
on_monthly_pulse = {
	if = {
		limit = { hel_is_confederation = yes }
		if = {
			limit = { is_at_war = yes }
			clr_country_flag = hel_peace_since
		}
		else_if = {
			limit = { NOT = { has_country_flag = hel_peace_since } }
			set_country_flag = hel_peace_since
		}
	}
}
